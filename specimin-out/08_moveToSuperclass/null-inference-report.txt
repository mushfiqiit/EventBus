class SubscriberMethodFinder {

        @Nullable
        Class<?> clazz;

        void moveToSuperclass() {
            if (skipSuperClasses) {
                clazz = null;
            } else {
                clazz = clazz.getSuperclass();
                if (clazz != null) {

                    if (clazzName.startsWith("java.") || clazzName.startsWith("javax.") || clazzName.startsWith("android.") || clazzName.startsWith("androidx.")) {
                        clazz = null;
                    }
                }
            }
        }
    }
}
